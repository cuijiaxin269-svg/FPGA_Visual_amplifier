from pathlib import Path
import sys,json
import numpy as np
ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT/'algorithm'))
from motion60_spectrum import MotionSpectrum,SpectrumConfig
folder=ROOT/'build/roi256_vectors';folder.mkdir(parents=True,exist_ok=True)
rng=np.random.default_rng(7100)
base=rng.uniform(100,900,(256,256))
transform=np.fft.fft2(base)
fy=np.fft.fftfreq(256)[:,None];fx=np.fft.fftfreq(256)[None,:]
model=MotionSpectrum(SpectrumConfig())
shifts=[(0,0),(.4,-.3),(2.3,1.7),(-3.2,-1.2)]
expected=[]
for i,(dx,dy) in enumerate(shifts):
    roi=np.fft.ifft2(transform*np.exp(-2j*np.pi*(fx*dx+fy*dy))).real
    frame=np.zeros((800,1280),dtype='<u2')
    frame[272:528,512:768]=np.clip(np.rint(roi),0,1023).astype('<u2')
    frame.tofile(folder/f'input{i}.bin')
    r=model.push(frame,i)
    expected.append([r['dx_px'],r['dy_px'],r['correlation_peak'],r['variance_dn2']])
np.array(expected,dtype='<f4').tofile(folder/'expected.bin')
(folder/'metadata.json').write_text(json.dumps(dict(shifts=shifts,expected=expected,maximum_displacement_error_px=.002,maximum_peak_error=.002),indent=2)+'\n')
print(folder)
