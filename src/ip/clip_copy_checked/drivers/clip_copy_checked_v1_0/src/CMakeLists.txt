cmake_minimum_required(VERSION 3.15)
project(clip_copy_checked)
find_package(common)
collector_create (PROJECT_LIB_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}")
collect (PROJECT_LIB_SOURCES xclip_copy_checked.c)
collect (PROJECT_LIB_SOURCES xclip_copy_checked_sinit.c)
collect (PROJECT_LIB_SOURCES xclip_copy_checked_linux.c)
collect (PROJECT_LIB_SOURCES xclip_copy_checked_g.c)
collector_list (_sources PROJECT_LIB_SOURCES)
collector_create (PROJECT_LIB_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}")
collect (PROJECT_LIB_HEADERS xclip_copy_checked.h)
collect (PROJECT_LIB_HEADERS xclip_copy_checked_hw.h)
collector_list (_headers PROJECT_LIB_HEADERS)
file(COPY ${_headers} DESTINATION ${CMAKE_BINARY_DIR}/include)
include_directories(${CMAKE_BINARY_DIR}/include)
add_library(clip_copy_checked STATIC ${_sources})
set_target_properties(clip_copy_checked PROPERTIES LINKER_LANGUAGE C)
